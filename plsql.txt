DECLARE
    CURSOR guest_cursor IS
        SELECT first_name, phone
        FROM guests
        WHERE first_name LIKE 'A%';
BEGIN
    FOR guest_rec IN guest_cursor LOOP
        DBMS_OUTPUT.PUT_LINE('Name: ' || guest_rec.first_name || ', Phone: ' || guest_rec.phone);
    END LOOP;
END;
